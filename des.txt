users
PK: id
Columns: id, username, password, role, full_name, created_at
customers
PK: id
Columns: id, full_name, phone, points
genres
PK: id
Columns: id, name
movies
PK: id
Columns: id, title, duration, status, poster_url, deleted_at
movie_genres (bảng trung gian many-to-many giữa phim và thể loại)
PK: (movie_id, genre_id)
Columns: movie_id, genre_id
FK: movie_genres.movie_id → movies.id, movie_genres.genre_id → genres.id
rooms
PK: id
Columns: id, name, type
seats
PK: id
Columns: id, room_id, row_char, number, type
FK: seats.room_id → rooms.id
showtimes
PK: id
Columns: id, movie_id, room_id, start_time, end_time, base_price
FK: showtimes.movie_id → movies.id, showtimes.room_id → rooms.id
products
PK: id
Columns: id, name, price, unit, deleted_at
inventories
PK: product_id
Columns: product_id, current_quantity, min_threshold
FK: inventories.product_id → products.id

stock_receipts
PK: id
Columns: id, receipt_date, supplier, total_import_cost
stock_receipt_items
PK: id
Columns: id, receipt_id, product_id, quantity, import_price
FK: stock_receipt_items.receipt_id → stock_receipts.id, stock_receipt_items.product_id → products.id
promotions
PK: id
Columns: id, code, discount_percent
invoices (Master của giao dịch bán vé)
PK: id
Columns: id, user_id, customer_id, promotion_id, created_at, status, final_total
FK: invoices.user_id → users.id, invoices.customer_id → customers.id, invoices.promotion_id → promotions.id
tickets (Detail 1 của invoice)
PK: id
Columns: id, invoice_id, showtime_id, seat_id, final_price
FK: tickets.invoice_id → invoices.id, tickets.showtime_id → showtimes.id, tickets.seat_id → seats.id
invoice_items (Detail 2 của invoice: đồ ăn/đồ uống)
PK: id
Columns: id, invoice_id, product_id, quantity, unit_price
FK: invoice_items.invoice_id → invoices.id, invoice_items.product_id → products.id
payments
PK: id
Columns: id, invoice_id (UNIQUE), amount, method, transaction_id, status
FK: payments.invoice_id → invoices.id
